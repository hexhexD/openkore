sub sorting_numbers {
	my @nums = ( $_[0] =~ /(\d+)/g );
	message("eventmacro sorting numbers\n");
	my @sorted = sort { substr($a, -1) <=> substr($b, -1) } @nums ;
	message("\@sorted: @sorted\n");
	my @result = map { substr($_, 0, 1) } @sorted;
	message("\@result: @result\n");
	return @result;
}
