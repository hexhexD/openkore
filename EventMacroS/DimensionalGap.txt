automacro ghost_boss_death {
	InMap 1@spa
	SimpleHookEvent target_died
	call {
		if ($.SimpleHookEventLastMonster =~ /呪いの根源/) {
			[
			do conf itemsGatherAuto 2
      do conf itemsTakeAuto 2
			do conf route_randomWalk 0
			]
			log 呪いの根源 dead xD
			do move 202 38
			do talknpc 196 44 r0 r0 r0
			do move 196 44

			# walk to make sure we pick up stuff
			do move 198 27
			do move 196 44
			# exit
			do move 198 27
			do talknpc 204 29 r0 r0
			do move 211 27
			do conf itemsGatherAuto 0

			MD(ghost)
		}
	}
}
