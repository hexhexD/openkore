automacro gx_edp {
	MobNear /(フェンリル|呪いの根源)/i
	StatusInactiveHandle EFST_EDP
	disabled 1
	timeout 5
	overrideAI 1
	exclusive 1
	call {
		do ss ASC_EDP
	}
}
