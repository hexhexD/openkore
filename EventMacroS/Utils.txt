# Store euqipments and cycle to next character
macro cycle {
  set exclusive 1
  do em enable mdStart

  $char = &config(char)
  $TOTAL = 4

  log current char is $char

  $char = &eval(($char + 1) % $TOTAL)
  if (&defined($.param[0])) {
    $char = $.param[0]
  }

  log new char is $char
  do conf char $char

  $char = unset
  $TOTAL = unset

  call gxstore

  do charselect

  call gxget
}

macro airseiren_all {

  log 精錬 $sei_name

    log No +0 item found
    $seiren_total = 0
  } else {
    $seiren_total = @sei_base

  log Number of 0 refinement: $seiren_total
}

macro get_storage_eq_num {
  $result = &Storage($.param[0])
  @eq = &split(',', $result)
  $j = 0
  [
  while ($j < $.param[1]) {
    log $eq[$j]
    do storage get $eq[$j]
    $j++
  }
  ]
}
macro store_storage_eq_num {
  $result = &Inventory($.param[0])
  @eq = &split(',', $result)
  $j = 0
  [
  while ($j < $.param[1]) {
    log $eq[$j]
    do storage add $eq[$j]
    $j++
  }
  ]
}
