 macro regroup {
	do party leave
	$number = &rand(666, 999)
	$name = card お願いします$number
	do party create $name
[
	do conf itemsTakeAuto 0
	do conf itemsGatherAuto 0
	do conf attackAuto 0
	do conf route_randomWalk 0
]
}
