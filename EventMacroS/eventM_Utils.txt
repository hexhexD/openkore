	$airship_dst = Rachel

}

macro rachel_yuno {
    set exclusive 1
	$airship_dst = yuno
	do move ra_fild12 295 208
	do talk resp 0
}

automacro getoff_yuno {
    LocalMsg /現在、ジュノーに停船中です/
    call {
		log Getting off at Yuno
		if ($airship_dst == yuno) {
			do move 242 72
		}
    }
}
automacro getoff_rachel {
    LocalMsg /現在、ラヘルに停船中です/
    call {
		log Getting off at Rachel
		if ($airship_dst == Rachel) {
			do move 242 72
			do move rachel
		}
    }
}
