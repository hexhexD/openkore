# refine all the same euipment, starting from zero until + 7
# e.g. em seiren -- "メロンワンド [1]" Don't forget the quote!!
macro seiren {
[
  $sei_name = $.param[0]
  log 精錬 $sei_name
  @sei_base = &Inventory($sei_name)
  if ($sei_base[0] == -1) {
    log No +0 item found
    $seiren_total = 0
  } else {
    $seiren_total = @sei_base
  }
  log Number of 0 refinement: $seiren_total

  $seiren_i = 0
  while ($seiren_i < $seiren_total) {
    log Refining index $sei_base[$seiren_i]
    call seiren_work $sei_base[$seiren_i] 0
    $seiren_i++
  }

  $sei_plus4name = +4 $sei_name
  @sei_plus4 = &Inventory($sei_plus4name)
  if ($sei_plus4[0] == -1) {
    log No $sei_plus4name found
    $seiren_total = 0
  } else {
    $seiren_total = @sei_plus4
  }
  $seiren_i = 0
  while ($seiren_i < $seiren_total) {
    log Refining index $sei_plus4[$seiren_i]
    call seiren_work $sei_plus4[$seiren_i] 4
    $seiren_i++
  }

  $sei_plus5name = +5 $sei_name
  @sei_plus5 = &Inventory($sei_plus5name)
  if ($sei_plus5[0] == -1) {
    log No $sei_plus5name found
    $seiren_total = 0
  } else {
    $seiren_total = @sei_plus5
  }
  $seiren_i = 0
  while ($seiren_i < $seiren_total) {
    log Refining index $sei_plus5[$seiren_i]
    call seiren_work $sei_plus5[$seiren_i] 5
    $seiren_i++
  }

  $sei_plus6name = +6 $sei_name
  @sei_plus6 = &Inventory($sei_plus6name)
  if ($sei_plus6[0] == -1) {
    log No $sei_plus6name found
    $seiren_total = 0
  } else {
    $seiren_total = @sei_plus6
  }
  $seiren_i = 0
  while ($seiren_i < $seiren_total) {
    log Refining index $sei_plus6[$seiren_i]
    call seiren_work $sei_plus6[$seiren_i] 6
    $seiren_i++
  }
]
}

# melon normal refine guy
macro seiren_work {
  do eq $.param[0]
  if ($.param[1] < 4) {
    # instant + 4
    do talknpc 105 61 r0 r3 r1 r0
  } else {
    do talknpc 105 61 r0 r3 r0 r0
  }

}

# "store" means store, "get" means get
macro gx_rearm { 
  do uneq all
  do is ペコペコの手綱

  @equipmentList = undef;
  &push(@equipmentList, [衣装] 輝くマジェスティックゴート);
  &push(@equipmentList, +9 黎明のマジェスティックゴート [Extra Str*2:MaxHP + 15%:固定詠唱時間 - 50%]);
  &push(@equipmentList, +9 フェイスオブイグドラシル [厄災の魔将:英雄の凱歌] [1]);
  &push(@equipmentList, くわえたチョコスティック);
  &push(@equipmentList, +9 メロンカタール [Special Str:アルカナ:覇王] [1]);
  &push(@equipmentList, +9 ヴァルキリーカースブライト [ニヒツ:覇王] [1]);
  &push(@equipmentList, +9 デスブリンガー [スキルディレイ - 15%:フラッド] [1]);
  &push(@equipmentList, 銀の矢);
  &push(@equipmentList, +9 銀の毛皮のシューズ [Extra Str:MaxHP + 15%:スキルディレイ - 15%*2]);
  &push(@equipmentList, リングオブシルバークロウ [Special Luk:Special Str:Vit + 7] [1]);
  &push(@equipmentList, カルニウムイヤリング [Extra Str:MaxHP + 15%:スキルディレイ - 15%:詠唱時間 - 15%]);
  &push(@equipmentList, ペコペコの手綱);
  &push(@equipmentList, 怨嗟の腕輪);
  &push(@equipmentList, 通行証);
  log Equipment list size is @equipmentList

  do move yuno 195 306
  # Main storage
  do talknpc 199 299

  set macro_delay 0.1
  if ($.param[0] == "store") {
    log Storing equipments
    $i = 0
    while ($i < @equipmentList) {
      log Storing $equipmentList[$i]
      do storage add $equipmentList[$i] 1
      $i++
    }
  } elsif ($.param[0] == "get") {
    $i = 0
    while ($i < @equipmentList) {
      log Getting $equipmentList[$i]
      do storage get $equipmentList[$i] 1
      $i++
    }
  }
  # Pause so we don't immediately close storage after blasting the server with all store/get all at once
  do storage close
}
