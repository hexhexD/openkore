automacro HIT_training {
    NpcNear /パワースポット/
    timeout 1
    disabled 1
    call {
		log $.NpcNearLastPos
		$x = &arg ("$.NpcNearLastPos", 1)
		$y = &arg ("$.NpcNearLastPos", 2)
		do talknpc $x $y c
    }
}

}
macro rachel_yuno {

}
automacro getoff_yuno {
    LocalMsg /現在、ジュノーに停船中です/
    call {
		log Getting off at Yuno
		if ($airship_dst == yuno) {
			do move 242 72
		}
    }
}
automacro getoff_rachel {
    LocalMsg /現在、ラヘルに停船中です/
    call {
		log Getting off at Rachel
		if ($airship_dst == Rachel) {
			do move 242 72
			do move rachel
		}
    }
}
